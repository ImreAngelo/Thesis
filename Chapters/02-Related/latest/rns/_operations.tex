
\subsection{RNS Base Conversion}
In later sections we will need to efficiently express a polynomial in different \textit{bases} i.e. moduli.

\begin{definition}[RNS Basis]
    Given a large modulus $Q = q_0 q_1\dots q_{k-1}$, the \textit{RNS basis} $\mathcal{B}_Q = \{q_0, q_1, \dots, q_{k-1}\}$ denotes the moduli used in the RNS representation of $a\in R_Q$. A polynomial $a \in R_Q$ can then be expressed by its residues modulo the elements of $\mathcal{B}_Q$

    \begin{equation}
        \label{def:rns-base}
        a\in \mathcal{B}_Q \implies a \equiv r_0 \pmod{q_0}, ...
    \end{equation}
    
    % We can denote an RNS polynomial $a \in R_Q$ as $a \in \mathcal{B}_Q$. (\textcolor{red}{RNS = CRT \textit{or} DCRT})
    % The basis $\mathcal{B}_Q = \{q_0, q_1, \dots, q_{k-1}\}$ of the big modulus is just a collection of the smaller (pairwise coprime) RNS primes $q_0\dots q_{k-1} = Q$. By some abuse of notation we can denote an RNS (CRT or DCRT) polynomial $a \in R_Q$ as $a \in \mathcal{B}_Q$.
\end{definition}

\begin{remark}
    % While all these forms ultimately represent the same polynomial, converting between them is not trivial. Later sections will go into detail on when and how to convert between different representations of the same polynomial. 
    To avoid confusion, I will denote a polynomial $a$ in \textit{coefficient mode} \eqref{form:coefficient} as an element in the ring $a \in R_Q$ and a polynomial in \textit{NTT mode} as an element of the base $a \in \mathcal{B}_Q$. If a polynomial is expressed in \textit{CRT mode} \eqref{form:crt}, then it will be explicitly stated.
\end{remark}

\subsubsection{Base Extension}
We have a polynomial $a \in \mathcal{B}_Q$ in evaluation form \eqref{form:eval} and want to express the same polynomial in $\mathcal{B}_{QP}$, where $Q = q_0\dots q_k$ and $P = p_0\dots p_m$. $P$ is chosen such that $\gcd(p_i, p_j) = 1$ for all $i\not= j$ and $\gcd(p_i, g_j) = 1$ for all $p_i, g_j$ (as well as $\gcd(q_i,q_j) = 1 \forall i\not= j$). As $Q$ and $P$ are disjoint, $QP$ shares the $q$ residues so we do not need to re-calculate them.

%%%
%%% Base Extension
%%%
\begin{algorithm}[ht!]
\caption{Base Extension}
\label{alg:base_extension}
\begin{algorithmic}[1]
    \Require $a \in \mathcal{B}_Q$, $\mathcal{B}_P$
    \Ensure $a \in \mathcal{B}_{QP}$
    \State Initialize an empty array $a_{P}$ of size $\|\mathcal{B}_P\|$
    \Comment{Holds the $P$-towers}
    \State $a_{\mathsf{coeffs}} := \mathsf{Inverse CRT}(\mathsf{Inverse NTT}(a))$
    \Comment{Convert to coefficient form}
    \For{$p_i \in \mathcal{B}_P$}
        \State $a_{p_i} := a_{\mathsf{coeffs}} \pmod{p_i}$ \Comment{Find $p_i$-tower}
        \State Append $\mathsf{NTT}(a_{p_i})$ to $a_{P}$
    \EndFor
    \State\Return $a \cup a_{P}$
\end{algorithmic}
\end{algorithm}

\subsubsection{Fast Base Extension}
The base extension requires converting an NTT polynomial $a \in \mathcal{B}_Q$ to CRT mode, then coefficient mode. 
% an INTT to convert $a$ to coefficient form, CRT, then inverse CRT and an NTT to convert $a\bmod{p_i}$ to evaluation form. 
The improvements made by \cite{cryptoeprint:2016/510} performs the base extension in the CRT representation, saving one inverse CRT at the cost of some additional noise.

\begin{equation}
    \label{eq:fastbconv}
    \mathsf{FastBConv(x,Q, B)} = \left(\sum_{i=1}^{k}\left|x_i\frac{q_i}{Q}\right|_{q_i} \times \frac{Q}{q_i}\right)_{m\in B}
\end{equation}

%%%
%%% Fast Base Extension
%%%
\begin{algorithm}[ht!]
\caption{Fast Base Extension}
\label{alg:fast_base_extension}
\begin{algorithmic}[1]
    \Require $a \in \mathcal{B}_Q$, $\mathcal{B}_P$
    \Ensure $(a + \epsilon_P) \in \mathcal{B}_{QP}$, where $\epsilon_P = \alpha_xQ$ for some $\alpha_x\in[0, k)$
    \State $a_{Q} := \mathsf{Inverse NTT}(a)$
    \Comment{Convert to CRT form}
    \State $a_{P} := \mathsf{FastBConv(a_Q, \mathcal{B}_Q, \mathcal{B}_P)}$
    \State\Return $a \cup \mathsf{NTT}(a_{P})$
\end{algorithmic}
\end{algorithm}

\begin{remark}
    Note that the factors $|\frac{q_i}{Q}|_{q_i}$ and $\frac{Q}{q_i}$ can be pre-calculated and stored in memory (will be discussed in \autoref{sec:impl:hps}).
\end{remark}

The fast base extension calculates $x + \alpha_x q$ for some integer $\alpha_x \in [0, k-1]$. The paper also shows how to correct this error...

\begin{algorithm}[ht!]
\caption{Fast Base Extension $Q \to QP$ (ApproxModUp)}
\label{alg:fast_base_extension_2}
\begin{algorithmic}[1]
\Require $x = \{x_i\}_{i=0}^{k}$, with $x_i = x \bmod q_i$, in base $Q=\{q_0,\dots,q_k\}$
\Require Precomputed $\tilde q_i = \left(\frac{Q}{q_i}\right)^{-1}\!\bmod q_i$ for $i=0,\dots,k$
\Require Precomputed $\omega_{i,\ell} = \frac{Q}{q_i}\bmod p_\ell$ (Shoup form) for $i=0,\dots,k$,\ $\ell=0,\dots,j$
\Ensure $x' = \{x'_s\}_{s \in QP}$, the residues of $x + uQ$ in base $QP=\{q_0,\dots,q_k,p_0,\dots,p_j\}$,\ $\|u\|_\infty < \tfrac{k+1}{2}$
\Statex
\State \textbf{// shared source-side residues (hoisted, target-independent)}
\For{$i = 0$ \textbf{to} $k$}
    \State $y_i \gets x_i \cdot \tilde q_i \bmod q_i$
\EndFor
\Statex
\State \textbf{// $Q$-part: reuse input limbs, zero cost, overflow-free}
\For{$i = 0$ \textbf{to} $k$}
    \State $x'_{q_i} \gets x_i$
\EndFor
\Statex
\State \textbf{// $P$-part: the only arithmetic, with lazy reduction}
\For{$\ell = 0$ \textbf{to} $j$}
    \State $\mathrm{acc} \gets 0$ \Comment{wide / unreduced accumulator}
    \For{$i = 0$ \textbf{to} $k$}
        \State $\mathrm{acc} \gets \mathrm{acc} + y_i \cdot \omega_{i,\ell}$ \Comment{ShoupMulAdd, deferred reduction}
    \EndFor
    \State $x'_{p_\ell} \gets \mathrm{acc} \bmod p_\ell$
\EndFor
\Statex
\State \Return $x'$
\end{algorithmic}
\end{algorithm}
