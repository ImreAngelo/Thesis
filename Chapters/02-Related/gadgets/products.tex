\begin{definition}[External Product]
    \begin{align}
        \boxdot: \Call{RLWE}{\mu} &\boxdot \Call{RGSW}{\gamma} = \Call{RLWE}{\mu} \\
        x &\boxdot C = G^{-1}(x) \cdot C
    \end{align}
\end{definition}

\begin{lemma}[Gadget Matrix Construction]
    Two vectors $P(u), D(v)$ that satisfy the gadget property \eqref{def:gadget-property} define a valid pair of gadget matrices with $g=P(1)$:
    \begin{gather}
        \label{def:Gmat}
        G = I_2 \otimes P(1) \\
        \label{def:Ginvmat}
        G^{-1}(x) = [D(x_0)|D(x_1)] \\
        \text{s.t. } G^{-1}(x)\cdot G = x + \epsilon \pmod{Q}
    \end{gather}

    \begin{proof}
        \begin{gather*}
            G^{-1}(x)\cdot G = [D(x_0)|D(x_1)] 
            \begin{pmatrix}
                P(1) & 0 \\
                0 & P(1)
            \end{pmatrix} \\
            \left(
                D(x_0) \cdot P(1), D(x_1)\cdot P(1)
            \right)
            = (x_0 + \epsilon_0, x_1 + \epsilon_1) = x + \epsilon \pmod{Q}
        \end{gather*}
    \end{proof}
\end{lemma}

\begin{definition}[Internal Product]
    \begin{gather}
        \boxtimes: \Call{RGSW}{a}\boxtimes\Call{RGSW}{b} = \Call{RGSW}{a\cdot b} \\
        A \boxtimes C = \begin{bmatrix}
            A_0 \boxdot C \\
            A_1 \boxdot C \\
            \vdots \\
            A_{2\ell - 1} \boxdot C
        \end{bmatrix}
    \end{gather}
\end{definition}
