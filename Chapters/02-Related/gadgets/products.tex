\subsection{RGSW}
The gadget techniques above are used in the \gls{gsw} scheme~\cite{cryptoeprint:2013/340} and its ring variants~\cite{cryptoeprint:2014/816, cryptoeprint:2018/421}, where the ciphertext itself is built around a gadget. \gls{gsw} came with a homomorphic product between two of its ciphertexts, and TFHE later added an asymmetric product between an \gls{rlwe} and an \gls{rgsw} ciphertext~\cite{cryptoeprint:2016/870, cryptoeprint:2018/421}. This subsection defines the \gls{rgsw} ciphertext and both products.

\begin{definition}[Gadget Matrix]
    \label{def:gadget-matrix}
    A gadget $g \in \mathbb{Z}_Q^\ell$ is extended to $k$-dimensional ciphertexts by the block-diagonal gadget matrix $G = I_k \otimes g \in R_Q^{k\ell\times k}$. % \emph{gadget matrix}
    % \begin{equation}
    %     G = I_k \otimes g \in R_Q^{k\ell\times k}.
    % \end{equation}
    % \gls{rlwe} ciphertexts are pairs, so $G = I_2 \otimes g$ throughout this thesis.
\end{definition}

\begin{definition}[RGSW Ciphertext]
    Given a gadget matrix $G = I_2 \otimes g$, let $Z \in R_Q^{2\ell\times 2}$ be a matrix whose $2\ell$ rows are fresh encryptions $\Call{RLWE}{0}$. An \gls{rgsw} encryption of $\mu \in R_Q$ is
    \begin{equation}
        \label{def:rgsw}
        C = Z + \mu\cdot G \in R_Q^{2\ell\times 2}
    \end{equation}
\end{definition}

An \gls{rgsw} ciphertext encodes the gadget-scaled messages $\mu g_i$ row by row. In the top block, $\mu g_i$ is added to the masked element, replacing the zero message of the underlying $\Call{RLWE}{0}$; in the bottom block it is added to the public element, so the row decrypts to $-\mu g_i\cdot s$ instead. Every row of $C$ is therefore itself an \gls{rlwe} ciphertext.

Two \gls{rgsw} ciphertexts add homomorphically, just as \gls{rlwe} ciphertexts do: the zero encryptions and the gadget terms add component-wise, so $C_A + C_B$ encrypts $\mu_A + \mu_B$. Addition alone is, however, not sufficient for \gls{fhe}. The multiplication mirrors the vector--matrix relationship between the two ciphertext types: the \gls{rlwe} ciphertext $x = (x_0, x_1)$ is mapped by a two-dimensional decomposition $G^{-1}(x)$ to a row vector of $2\ell$ small ring elements, which multiplies the $2\ell\times 2$ matrix $C$. The only requirement on $G^{-1}(x)$ is that it approximately inverts the $G$ matrix: 
\begin{equation}
    \label{eq:gadget-matrix-property}
    G^{-1}(x)\cdot G = x + e \pmod{Q}
\end{equation}

with quality $\|G^{-1}(x)\| \leq \beta$. No particular decomposition is prescribed; any $G^{-1}(x)$ satisfying \eqref{eq:gadget-matrix-property} yields a valid product. % a generality that later chapters exploit by instantiating $G^{-1}$ with different gadgets. %% DO NOT FORWARD REFERENCE!

\begin{definition}[External Product]
    \label{def:prod:external}
    Let $x \in R_Q^2$ be an \gls{rlwe} encryption of $m \in R_t$, let $C \in R_Q^{2\ell\times 2}$ be an \gls{rgsw} encryption of $\mu \in R$ under the same secret key, and let $G^{-1}$ be a gadget decomposition satisfying \eqref{eq:gadget-matrix-property}. The {external product} is defined as
    \begin{align}
        \boxdot: \Call{RLWE}{} \times \Call{RGSW}{} &\longrightarrow \Call{RLWE}{} \nonumber \\
        \label{def:extprod}
        (x, C) &\longmapsto x \boxdot C = G^{-1}(x) \cdot C,
    \end{align}
    which returns an encryption of the product of the messages: $\Call{RLWE}{m} \boxdot \Call{RGSW}{\mu} = \Call{RLWE}{m\cdot\mu}$.
    Correctness follows by inserting \eqref{def:rgsw} into \eqref{def:extprod}:
    \begin{equation}
        \label{eq:extprod-expansion}
        G^{-1}(x)\cdot C = G^{-1}(x)\cdot Z + \mu\cdot(x + e)
    \end{equation}
    where $\mu\cdot x$ carries the message and the remaining terms contribute only noise; formal correctness and noise bounds are given in~\cite{cryptoeprint:2018/421}.
\end{definition}

\begin{lemma}[Gadget Matrix Construction]
    Two vectors $P(u), D(v)$ that satisfy the gadget property \eqref{def:gadget-property} define a valid pair of gadget matrices with $g=P(1)$:
    \begin{gather}
        \label{def:Gmat}
        G = I_2 \otimes P(1) \\
        \label{def:Ginvmat}
        G^{-1}(x) = [D(x_0)|D(x_1)]
    \end{gather}
    satisfying \eqref{eq:gadget-matrix-property}, with $\|G^{-1}(x)\| \leq \beta$ inherited from the quality of $D$.

    \begin{proof}
        The block structure reduces the product to one inner product per component, each an instance of \eqref{def:gadget-property}:
        \begin{gather*}
            G^{-1}(x)\cdot G = [D(x_0)|D(x_1)]
            \begin{pmatrix}
                P(1) & 0 \\
                0 & P(1)
            \end{pmatrix} \\
            \big(\langle D(x_0), P(1)\rangle,\; \langle D(x_1), P(1)\rangle\big)
            = (x_0 + e_0,\; x_1 + e_1) = x + e \pmod{Q}
        \end{gather*}
    \end{proof}
\end{lemma}

\begin{remark}%[Asymmetric Noise Growth]
    \label{rem:extprod-asymmetry}
    The two noise terms of \eqref{eq:extprod-expansion} behave very differently, as detailed in \autoref{appendix:asymmetry}. The term $G^{-1}(x)\cdot Z \leq (a\cdot s + \beta\sum_{i=0}^{2\ell-1} \epsilon_i,\; a)$ amplifies the masking errors of the \gls{rgsw} operand by the decomposition quality. The precision term $e$ is scaled by $\mu$, which is why $\mu$ is typically constrained to $\{0,1\}$ when an inexact gadget is used. The error of the \gls{rlwe} operand, by contrast, is never amplified: it is carried into the result scaled only by $\mu$. This asymmetry was observed early on~\cite{cryptoeprint:2013/541, cryptoeprint:2014/094, cryptoeprint:2014/816}, and makes \gls{gsw}-style ciphertexts particularly suited to evaluating long chains of products. The noise stays low as long as the right operand is fresh, or at least has a small $\|\epsilon\|_\infty$, regardless of the error already accumulated on the left.
\end{remark}

The original \gls{gsw} construction defined its homomorphic product directly between two of its ciphertexts~\cite{cryptoeprint:2013/340}. Since every row of an \gls{rgsw} ciphertext is itself an \gls{rlwe} ciphertext, this \emph{internal product} is recovered by applying the external product row by row.

\begin{definition}[Internal Product]
    \label{def:prod:internal}
    Let $A, C \in R_Q^{2\ell\times 2}$ be \gls{rgsw} encryptions of $\mu_A, \mu_C \in R$ under the same secret key and gadget, and let $A_i$ denote the $i$-th row of $A$. The \emph{internal product}~\cite{cryptoeprint:2018/421} applies the external product to each row of the left operand:
    \begin{align}
        \boxtimes: \Call{RGSW}{} \times \Call{RGSW}{} &\longrightarrow \Call{RGSW}{} \nonumber \\
        (A, C) &\longmapsto A \boxtimes C = \begin{bmatrix}
            A_0 \boxdot C \\
            A_1 \boxdot C \\
            \vdots \\
            A_{2\ell - 1} \boxdot C
        \end{bmatrix},
    \end{align}
    which returns an encryption of the product of the messages: $\Call{RGSW}{\mu_A} \boxtimes \Call{RGSW}{\mu_C} = \Call{RGSW}{\mu_A\cdot\mu_C}$.
\end{definition}

One internal product amounts to $2\ell$ external products, and correctness follows row by row: each external product multiplies the content of $A_i$ by $\mu_C$, so the stacked result is again of the form \eqref{def:rgsw} with message $\mu_A\cdot\mu_C$.

The asymmetry of \autoref{rem:extprod-asymmetry} applies to each of the $2\ell$ rows, and dictates the order in which a chain of products should be evaluated: the noise of the right-hand operand is amplified by the decomposition in every row of the result, while the noise of the left-hand operand is only carried through. When accumulating a running product, the accumulator must therefore be kept on the left with each fresh ciphertext supplied on the right, so that the noise grows additively rather than multiplicatively.
