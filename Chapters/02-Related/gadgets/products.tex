\subsection{RGSW}
\begin{itemize}
    \item GSW introduced in \cite{cryptoeprint:2013/340}
    \item and its ring variant \cite{cryptoeprint:2014/816}
    \item two-dimensional gadget property
\end{itemize}

\begin{definition}[RGSW Ciphertext]
    Given a gadget vector $g$ 

    \begin{equation}
        \label{def:rgsw}
        C = Z + \mu\cdot G
    \end{equation}
\end{definition}

\begin{itemize}
    \item We want a product between RLWE and RGSW ciphertexts
    \item The RLWE-RGSW relationship is similar to vector vs matrix
\end{itemize}

\begin{definition}[External Product]
    \begin{align}
        \boxdot: \Call{RLWE}{\mu} &\boxdot \Call{RGSW}{\gamma} = \Call{RLWE}{\mu} \\
        x &\boxdot C = G^{-1}(x) \cdot C
    \end{align}
\end{definition}

\begin{lemma}[Gadget Matrix Construction]
    Two vectors $P(u), D(v)$ that satisfy the gadget property \eqref{def:gadget-property} define a valid pair of gadget matrices with $g=P(1)$:
    \begin{gather}
        \label{def:Gmat}
        G = I_2 \otimes P(1) \\
        \label{def:Ginvmat}
        G^{-1}(x) = [D(x_0)|D(x_1)]
    \end{gather}
    satisfying \eqref{label}.

    \begin{proof}
        \begin{gather*}
            G^{-1}(x)\cdot G = [D(x_0)|D(x_1)] 
            \begin{pmatrix}
                P(1) & 0 \\
                0 & P(1)
            \end{pmatrix} \\
            \left(
                D(x_0) \cdot P(1), D(x_1)\cdot P(1)
            \right)
            = (x_0 + \epsilon_0, x_1 + \epsilon_1) = x + \epsilon \pmod{Q}
        \end{gather*}
    \end{proof}
\end{lemma}

\begin{remark}%[Asymmetric Noise Growth]
    \label{rem:extprod-asymmetry}
    The external product has two sources of noise, as detailed in \autoref{appendix:asymmetry}. The precision $\epsilon$ of the gadget decomposition is scaled by $\mu$. This is why $\mu$ is typically constrained to $\{0,1\}$ when an inexact gadget is used.
    The other source is the term $G^{-1}(x)\cdot Z \leq (a\cdot s + \beta\sum_{i=0}^{2\ell-1} \epsilon_i, a)$ which scales the sum of the masking errors by the decomposition quality. Consequently, the error of the left-hand-side has no effect on the error of the external product. This gives rise to an asymmetry where the external product has excellent noise-growth properties so long as the right hand side is always a fresh ciphertext/has small noise. % $\implies \epsilon \leq \beta\sum_i^{2\ell - 1} e_i \leq \beta\max_i\|e_i\|$

    %% TODO: Move derivation to appendix? 
\end{remark}

\begin{itemize}
    \item We also need a product definition between two \gls{rgsw} ciphertexts
    \item Using the fact that an RGSW ciphertext is a matrix of RLWE ciphertexts we can repeate the external product on each row
\end{itemize}

\begin{definition}[Internal Product]
    \begin{gather}
        \boxtimes: \Call{RGSW}{a}\boxtimes\Call{RGSW}{b} = \Call{RGSW}{a\cdot b} \\
        A \boxtimes C = \begin{bmatrix}
            A_0 \boxdot C \\
            A_1 \boxdot C \\
            \vdots \\
            A_{2\ell - 1} \boxdot C
        \end{bmatrix}
    \end{gather}
\end{definition}
