\subsection{Theoretical Hybrid Improvement}
\label{sec:results:hybrid}
An alternative to gadget decomposition often employed in \gls{rns} key-switching is \textit{hybrid} keyswitching \cite{cryptoeprint:2012/099, cryptoeprint:2021/204}. 

\begin{itemize}
    \item This approach nearly fits inside the gadget decomposition framework established in \autoref{sec:gadget-decomposition}, as $\langle u \cdot  P^{-1}, v\cdot P\rangle = u\cdot v \cdot P^{-1}\cdot P = u\cdot v \bmod{QP}$
    \item This actually defines a valid external product, the problem is that the $P$-scaling is consumed after this operation
\end{itemize}

\begin{definition}[Hybrid-RGSW Ciphertext]
    An \gls{rgsw} ciphertext that encodes $\mu P$ in the extended  
    \begin{equation}
        C_{QP} = Z_{QP} + \mu \begin{pmatrix}
            P & 0 \\
            0 & P
        \end{pmatrix} \pmod{QP}
    \end{equation}
    \begin{equation}
        \Call{HybridRGSW}{b} = \Call{RGSW}{b\cdot P}
    \end{equation}
\end{definition}

\begin{definition}[Hybrid Internal Product]
    I define the \textit{hybrid internal product} as 
    \begin{gather}
        \boxtimes_H \;:=\; \Call{RGSW}{a} \boxtimes_H \Call{RGSW}{b\cdot P} = \Call{RGSW}{a\cdot b}
    \end{gather}
\end{definition}

\begin{conjecture}[Mixed Internal Product]
    Performance $\mathcal{O}(k^2\ell/d_{num})$ instead of $\mathcal{O}(k^2\ell^2)$ because hybrid has $k/d_num$ rows.
    Is better for large $\ell$
\end{conjecture}
