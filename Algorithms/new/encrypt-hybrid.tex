\begin{algorithm}
\caption{Hybrid-RGSW Encryption}
\label{alg:hybrid:encrypt}
\begin{algorithmic}[1]
\Require Extended basis $\mathcal{B}_{P} = \{ p_0, \dots, p_{k/d_{num} - 1} \}$
\Require $\Call{NTT}{[\mu]}_{\mathcal{B}_Q} \in \mathbb{Z}_{q_0} \times \dots \times  \mathbb{Z}_{q_{k-1}}$
\Ensure $\Call{HGSW}{\mu}$
\State Initialize empty array $C$ with $2d_{num}$ slots
\State $m \gets \Call{Power}{\mu, P}$
\For{$r \gets [0, d_{num})$ \textbf{in parallel}}
    \State $mg \gets [m]_{\mathcal{B}_r}$
    \Comment{Zero every limb outside the block}
    \State $C_r \gets \Call{EncryptQP}{0} + (mg, 0) $
    \State $C_{r + d_{num}} \gets \Call{EncryptQP}{0} + (0, mg)$
\EndFor
\State \Return $C$
\end{algorithmic}
\end{algorithm}
